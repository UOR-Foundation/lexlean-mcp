definition{
  name: answerWith,
  parameters: [asked : StepServe::Asked],
  result: (Result Methods::ResultBody Refusal::ProviderRefusal),
  body: (match (Tools::admissionRefusal (GatewayTools::gatewayTools))
    | Option.some refusal => (ctor Result.error [Methods::ResultBody, Refusal::ProviderRefusal] refusal)
    | Option.none => (match (project asked request)
      | Methods::ClientRequest.toolsList params => (Tools::answerList (gatewayCache) (GatewayTools::gatewayTools) params)
      | Methods::ClientRequest.toolsCall params => (Tools::answerCall (GatewayTools::gatewayTools) (fun [asked] (call : Methods::CallToolParams) => (GatewayTools::behave (project asked completed) call)) params)
      | Methods::ClientRequest.serverDiscover params => (toolsOnly)
      | Methods::ClientRequest.resourcesList params => (toolsOnly)
      | Methods::ClientRequest.resourcesRead params => (toolsOnly)
      | Methods::ClientRequest.resourcesTemplatesList params => (toolsOnly)
      | Methods::ClientRequest.promptsList params => (toolsOnly)
      | Methods::ClientRequest.promptsGet params => (toolsOnly)
      | Methods::ClientRequest.completionComplete params => (toolsOnly)
      | Methods::ClientRequest.tasksGet params => (toolsOnly)
      | Methods::ClientRequest.tasksUpdate params => (toolsOnly)
      | Methods::ClientRequest.tasksCancel params => (toolsOnly))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

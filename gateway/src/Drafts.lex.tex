    | DraftAdmission::Verdict.Refused reason => (refusal (refusalName reason))

definition{
  name: renderedStep,

    others : (List (Bytes * Bytes)),

    outcome : Bytes,
    stored : (Option Bytes)

  result: Move,
  body: (match (Frame::u32At outcome 12)
    | Option.none => (refusal "OutcomeMalformed")
    | Option.some class => (match (Frame::chunkAt outcome 16)
      | Option.none => (refusal "OutcomeMalformed")
      | Option.some chunk => (if (class =? (classAccepted)) then (blobStep edit old others parts (project chunk bytes) stored) else (if (class =? (classRefused)) then (moduleRefused (project chunk bytes)) else (refusal "OutcomeMalformed"))))),

definition{
  name: renderStep,
  parameters: [
    edit : Edit,
    old : (Option String),
    others : (List (Bytes * Bytes)),
    parts : Record,
    seen : Seen
  ],
  result: Move,
  body: (match (DraftAdmission::admitDraft (maxFiles) (maxBytes) (project parts roots) (cons ((prim utf8_encode (project edit path) : Bytes), (JsonText::emptyBytes)) others))
    | DraftAdmission::Verdict.Refused reason => (refusal (refusalName reason))
    | DraftAdmission::Verdict.Admitted => (match (requestOf edit old)
      | Option.none => (ctor Move.Invalid "kind and header are needed to assemble a module")
      | Option.some request => (match (JsonWrite::writeJson request)
        | Result.error why => (ctor Move.Invalid "the data cannot be written as JSON")
        | Result.ok bytes => (match (project seen render)
          | Option.none => (ctor Move.Ask (storePutTag) (putPayloadAs (stagingKey (project edit profile) bytes) bytes (match old
            | Option.none => (formAssemble)
            | Option.some text => (formRewrite))))
          | Option.some outcome => (renderedStep edit old others parts outcome (project seen stored)))))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: rowsStep,
  parameters: [edit : Edit, rows : (List (Bytes * Bytes)), parts : Record, seen : Seen],
  result: Move,
  body: (let target : Bytes := (prim utf8_encode (project edit path) : Bytes); (let others : (List (Bytes * Bytes)) := (withoutPath rows target (nil (Bytes * Bytes))); (match (modeOf edit)
    | Mode.Revise => (match (findRow rows target)
      | Option.none => (refusal "PathUnknown")
      | Option.some held => (match (prim utf8_decode held : (Option String))
        | Option.none => (refusal "DraftCorrupt")
        | Option.some text => (renderStep edit (ctor Option.some [String] text) others parts seen)))
    | Mode.Compose => (renderStep edit (ctor Option.none [String]) others parts seen)
    | Mode.Wrong => (ctor Move.Invalid "kind and header are given together or not at all")))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: write,
  parameters: [
    profile : String,
    files : Json::Json,
    lexicon : (Option Bytes),
    stored : (Option Bytes)
  ],
  result: Move,
  body: (match (rowsOf files)
    | Option.none => (ctor Move.Invalid "files is not a list of objects that each have a text path and a text content")
    | Option.some rows => (match lexicon
      | Option.none => (ctor Move.Ask (lexiconTag) (lexiconPayload profile))
      | Option.some outcome => (match (foundContent outcome)
        | Option.none => (refusal "ProfileRecordMalformed")
        | Option.some entry => (match (recordOf entry)
          | Option.none => (refusal "ProfileRecordMalformed")
          | Option.some parts => (match (DraftAdmission::admitDraft (maxFiles) (maxBytes) (project parts roots) rows)
            | DraftAdmission::Verdict.Refused reason => (refusal (refusalName reason))
            | DraftAdmission::Verdict.Admitted => (let blob : Bytes := (preimage (project parts profile) rows); (let ident : String := (digestText blob); (match stored
              | Option.none => (ctor Move.Ask (storePutTag) (putPayload (handleOf profile ident) blob))
              | Option.some done => (written profile ident (DraftAdmission::countRows rows) (DraftAdmission::totalBytes rows)))))))))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

  body: (match (entriesOf rows)
    | Option.none => (refusal "DraftCorrupt")
    | Option.some entries => (ctor Move.Answer (prim join (entriesNames entries (nil String)) "\n" : String) (ctor Json::Json.object (cons ("files", (ctor Json::Json.array (entriesJson entries (nil Json::Json)))) (cons ("bytes", (ResultJson::natJson (DraftAdmission::totalBytes rows))) (nil (String * Json::Json))))) false)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: fileOf,
  parameters: [rows : (List (Bytes * Bytes)), path : String],
  result: Move,
  body: (match (findRow rows (prim utf8_encode path : Bytes))
    | Option.none => (refusal "PathUnknown")
    | Option.some content => (match (prim utf8_decode content : (Option String))
      | Option.none => (refusal "DraftCorrupt")
      | Option.some text => (ctor Move.Answer text (ctor Json::Json.object (cons ("path", (ctor Json::Json.text path)) (cons ("bytes", (ResultJson::natJson (prim length content : Nat))) (nil (String * Json::Json))))) false))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: readWith,
  parameters: [ident : String, want : Want, outcome : Bytes],
  result: Move,
  body: (match (foundContent outcome)
    | Option.none => (refusal "DraftUnknown")
    | Option.some blob => (if (prim equal (digestText blob) ident : Bool) then (match (decode blob)
      | Option.none => (refusal "DraftCorrupt")
      | Option.some rows => (match want
        | Want.Listing => (listed rows)
        | Want.File path => (fileOf rows path)
        | Want.Bad => (ctor Move.Invalid "path is not a string"))) else (refusal "DraftCorrupt"))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: readAt,
  parameters: [draft : String, want : Want, stored : (Option Bytes)],
  result: Move,
  body: (match (idOfHandle draft)
    | Option.none => (refusal "DraftUnknown")
    | Option.some ident => (match stored
      | Option.none => (ctor Move.Ask (storeGetTag) (getPayload draft))
      | Option.some outcome => (readWith ident want outcome))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: read,
  parameters: [draft : String, path : (Option Json::Json), stored : (Option Bytes)],
  result: Move,
  body: (match (wantOf path)
    | Want.Bad => (ctor Move.Invalid "path is not a string")
    | Want.Listing => (readAt draft (ctor Want.Listing) stored)
    | Want.File text => (readAt draft (ctor Want.File text) stored)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

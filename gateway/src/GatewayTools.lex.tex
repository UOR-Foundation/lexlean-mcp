definition{
  name: engineAnswer,
  parameters: [effect : Effects::Effect, outcome : Bytes],
  result: (Result Methods::Progressed Refusal::ProviderRefusal),
  body: (match (Frame::chunkAt outcome 8)
    | Option.none => (accepted (malformed))
    | Option.some chunk => (match (JsonRead::readJson (project chunk bytes))
      | Result.error refusal => (refused "the engine answered with bytes that are not JSON")
      | Result.ok document => (let class : Nat := (wordAt outcome 4); (if (if (prim equal (nameOf effect) "engine.snapshot" : Bool) then (if (class =? 0) then (isSnapshot document) else (exitCodeIs document class)) else (exitCodeIs document class)) then (accepted (reply (contentText (project chunk bytes)) document (if (class =? 0) then false else true))) else (refused "the engine answered with a document that is neither the snapshot nor the command result of its class"))))),

  executable: true

  body: (if ((project lead tag) =? (Drafts::lexiconTag)) then (if (failedWith (ctor Effects::Effect.LexiconRead) (project lead outcome)) then (accepted (hostFailure (ctor Effects::Effect.LexiconRead) (wordAt (project lead outcome) 4))) else (match rest
    | List.nil => (moved (Drafts::write profile files (ctor Option.some [Bytes] (project lead outcome)) (ctor Option.none [Bytes])))
    | List.cons later more => (match more
      | List.nil => (if ((project later tag) =? (Drafts::storePutTag)) then (if (failedWith (ctor Effects::Effect.StorePut) (project later outcome)) then (accepted (hostFailure (ctor Effects::Effect.StorePut) (wordAt (project later outcome) 4))) else (moved (Drafts::write profile files (ctor Option.some [Bytes] (project lead outcome)) (ctor Option.some [Bytes] (project later outcome))))) else (refused "the host performed another effect"))
      | List.cons again left => (refused "the host performed more than two effects")))) else (refused "the host performed another effect")),

  executable: true

definition{
  name: draftView,
  parameters: [
    completed : (List HostEffect::Completed),
    params : Methods::CallToolParams,
    draft : String,
    view : Nat

  body: (match (argValue params "path")
    | Option.none => (invalid "a data or spans view needs a path")
    | Option.some given => (match (Json::asText given)

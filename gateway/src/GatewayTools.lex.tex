definition{
  name: engineAnswer,
  parameters: [effect : Effects::Effect, outcome : Bytes],
  result: (Result Methods::Progressed Refusal::ProviderRefusal),
  body: (match (Frame::chunkAt outcome 8)
    | Option.none => (accepted (malformed))
    | Option.some chunk => (match (JsonRead::readJson (project chunk bytes))
      | Result.error refusal => (refused "the engine answered with bytes that are not JSON")
      | Result.ok document => (let class : Nat := (wordAt outcome 4); (if (if (prim equal (nameOf effect) "engine.snapshot" : Bool) then (if (class =? 0) then (isSnapshot document) else (exitCodeIs document class)) else (exitCodeIs document class)) then (accepted (reply (contentText (project chunk bytes)) document (if (class =? 0) then false else true))) else (refused "the engine answered with a document that is neither the snapshot nor the command result of its class"))))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: render,
  parameters: [effect : Effects::Effect, outcome : Bytes],
  result: (Result Methods::Progressed Refusal::ProviderRefusal),
  body: (if ((wordAt outcome 0) =? 1) then (accepted (hostFailure effect (wordAt outcome 4))) else (match effect
    | Effects::Effect.StoreGet => (accepted (foundReply outcome))
    | Effects::Effect.StorePut => (accepted (putReply outcome))
    | Effects::Effect.EngineCheck => (engineAnswer effect outcome)
    | Effects::Effect.EngineSnapshot => (engineAnswer effect outcome)
    | Effects::Effect.EngineFormat => (engineAnswer effect outcome)
    | Effects::Effect.EngineBuild => (engineAnswer effect outcome)
    | Effects::Effect.VerifyStart => (accepted (startReply outcome))
    | Effects::Effect.VerifyControl => (accepted (controlReply outcome))
    | Effects::Effect.LexiconRead => (accepted (foundReply outcome))
    | Effects::Effect.AttestationLookup => (accepted (foundReply outcome)))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true

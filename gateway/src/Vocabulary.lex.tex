definition{
  name: searchTool,
  parameters: [completed : (List HostEffect::Completed), params : Methods::CallToolParams],
  result: (Result Methods::Progressed Refusal::ProviderRefusal),
  body: (match (GatewayTools::argText params "profile")
    | Option.none => (GatewayTools::invalid "profile is not a string")
    | Option.some profile => (match (GatewayTools::optionalText params "query")
      | Option.none => (GatewayTools::invalid "query is not a string")

        | Option.none => (GatewayTools::invalid "package is not a string")

          | Option.none => (GatewayTools::invalid "offset is not a whole number")

            | Option.none => (GatewayTools::invalid "limit is not a whole number")

    | Option.none => (ctor Option.none [Wanted])

      | Option.none => (ctor Option.none [Wanted])

          | Option.none => (ctor Option.none [Wanted])
          | Option.some halves => (wantedFrom (first parts) (first halves) (second halves)))))),

  executable: true

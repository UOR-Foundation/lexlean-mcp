    | Alert.ClassTwo => (classFires (ctor ErrorClass.CliOrConfiguration) sample)
    | Alert.ClassThree => (classFires (ctor ErrorClass.Environment) sample)

      | Sample.ToolOutcome class count => false

      | Sample.QueueWait ms => false
      | Sample.ChildGroupEnds end count => (match end

        | GroupEnd.AlreadyGone => false

      | Sample.ProfileDrift profile count => false
      | Sample.SigningFailures count => false)
    | Alert.Drift => (match sample
      | Sample.ToolOutcome class count => false
      | Sample.ProtocolErrors code share => false
      | Sample.CheckLatency profile ms => false
      | Sample.VerifyDuration profile outcome ms => false
      | Sample.QueueWait ms => false
      | Sample.ChildGroupEnds end count => false

    | Alert.SigningFailure => (match sample
      | Sample.ToolOutcome class count => false

      | Sample.CheckLatency profile ms => false
      | Sample.VerifyDuration profile outcome ms => false
      | Sample.QueueWait ms => false
      | Sample.ChildGroupEnds end count => false
      | Sample.ProfileDrift profile count => false

definition{
  name: readHex,
  parameters: [count : Nat, accumulated : Nat, bytes : (List Nat)],
  result: Step,
  body: (match count
    | Nat.zero => (ctor Step.Got accumulated bytes)
    | Nat.succ previous => (match bytes
      | List.nil => (ctor Step.Stop (ctor Fault.Length))
      | List.cons head rest => (match (hexValue head)

definition{
  name: finish,
  parameters: [version : Nat, traceId : Nat, parentId : Nat, flags : Nat, rest : (List Nat)],
  result: Parent,
  body: (if (version =? 255) then (ctor Parent.Invalid (ctor Fault.ForbiddenVersion)) else (if (traceId =? 0) then (ctor Parent.Invalid (ctor Fault.ZeroTraceId)) else (if (parentId =? 0) then (ctor Parent.Invalid (ctor Fault.ZeroParentId)) else (match rest
    | List.nil => (ctor Parent.Valid (record TraceParent {version := version, traceId := traceId, parentId := parentId, flags := flags}))
    | List.cons head more => (if (version =? 0) then (ctor Parent.Invalid (ctor Fault.Length)) else (if (head =? 45) then (ctor Parent.Valid (record TraceParent {version := version, traceId := traceId, parentId := parentId, flags := flags})) else (ctor Parent.Invalid (ctor Fault.Separator))))))))

definition{

  parameters: [bytes : (List Nat)],
  result: Parent,
  body: (match (readHex 2 0 bytes)
    | Step.Stop fault => (ctor Parent.Invalid fault)
    | Step.Got version after1 => (match (readDash after1)
      | Step.Stop fault => (ctor Parent.Invalid fault)
      | Step.Got unused1 after2 => (match (readHex 32 0 after2)
        | Step.Stop fault => (ctor Parent.Invalid fault)
        | Step.Got traceId after3 => (match (readDash after3)
          | Step.Stop fault => (ctor Parent.Invalid fault)
          | Step.Got unused2 after4 => (match (readHex 16 0 after4)
            | Step.Stop fault => (ctor Parent.Invalid fault)
            | Step.Got parentId after5 => (match (readDash after5)
              | Step.Stop fault => (ctor Parent.Invalid fault)
              | Step.Got unused3 after6 => (match (readHex 2 0 after6)
                | Step.Stop fault => (ctor Parent.Invalid fault)

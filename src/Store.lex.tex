definition{
  name: sameOctets,
  parameters: [a : (List Nat), b : (List Nat)],
  result: Bool,
  body: (match a
    | List.nil => (match b | List.nil => true | List.cons y ys => false)
    | List.cons x xs => (match b
      | List.nil => false
      | List.cons y ys => (match (x =? y)
        | Bool.true => (sameOctets xs ys)
        | Bool.false => false))),
  recursive_argument: a
}

definition{
  name: idMatches,
  parameters: [stored : Sha256::Digest, found : Identity::Outcome],
  result: Bool,
  body: (match found
    | Identity::Outcome.Identified digest => (Identity::sameDigest digest stored)
    | Identity::Outcome.DuplicatePath => false
    | Identity::Outcome.DuplicateReview => false
    | Identity::Outcome.FieldTooLong => false)
}

definition{
  name: hashesMatch,
  parameters: [stored : (List (List Nat)), found : (List (List Nat))],
  result: Bool,
  body: (match stored
    | List.nil => (match found | List.nil => true | List.cons y ys => false)
    | List.cons x xs => (match found
      | List.nil => false
      | List.cons y ys => (match (sameOctets x y)
        | Bool.true => (hashesMatch xs ys)
        | Bool.false => false))),
  recursive_argument: stored
}

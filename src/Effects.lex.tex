definition{

  parameters: [d : Nat],
  result: (Result Effect Refusal),

    | Option.none => (ctor Result.error [Effect, Refusal] (wordRefusal 0 0 0 d)))
}

definition{
  name: decodeWord,
  parameters: [a : Nat, b : Nat, c : Nat, d : Nat],
  result: (Result Effect Refusal),
  body: (match a
    | Nat.zero => (match b
      | Nat.zero => (match c
        | Nat.zero => (decodeId d)
        | Nat.succ c1 => (ctor Result.error [Effect, Refusal] (wordRefusal a b c d)))
      | Nat.succ b1 => (ctor Result.error [Effect, Refusal] (wordRefusal a b c d)))
    | Nat.succ a1 => (ctor Result.error [Effect, Refusal] (wordRefusal a b c d)))
}

definition{
  name: decodeFrame,
  parameters: [octets : (List Nat)],
  result: (Result Effect Refusal),
  body: (match octets
    | List.nil => (ctor Result.error [Effect, Refusal] (ctor Refusal.MalformedFrame))
    | List.cons a r1 => (match r1
      | List.nil => (ctor Result.error [Effect, Refusal] (ctor Refusal.MalformedFrame))
      | List.cons b r2 => (match r2
        | List.nil => (ctor Result.error [Effect, Refusal] (ctor Refusal.MalformedFrame))
        | List.cons c r3 => (match r3
          | List.nil => (ctor Result.error [Effect, Refusal] (ctor Refusal.MalformedFrame))
          | List.cons d r4 => (match r4
            | List.nil => (decodeWord a b c d)
            | List.cons e r5 => (ctor Result.error [Effect, Refusal] (ctor Refusal.MalformedFrame)))))))
}

definition{
  name: octetsEq,
  parameters: [xs : (List Nat), ys : (List Nat)],
  result: Bool,
  body: (match xs
    | List.nil => (match ys | List.nil => true | List.cons y t => false)
    | List.cons x r => (match ys
      | List.nil => false
      | List.cons y t => ((prim equal x y : Bool) && (octetsEq r t)))),
  recursive_argument: xs
}

  name: readN,
  parameters: [count : Nat, bytes : (List Nat)],
  result: (Option ((List Nat) * (List Nat))),

    | Nat.zero => (ctor Option.some [((List Nat) * (List Nat))] ((nil Nat), bytes))

      | List.nil => (ctor Option.none [((List Nat) * (List Nat))])

  recursive_argument: count

  parameters: [left : (List Nat), right : (List Nat)],
  result: Bool,

    | List.nil => (match right | List.nil => true | List.cons y ys => false)

  recursive_argument: left

  name: compareOctets,
  parameters: [left : (List Nat), right : (List Nat)],
  result: Order,
  body: (match left

      | List.nil => (ctor Order.Equal)
      | List.cons y ys => (ctor Order.Less))
    | List.cons x xs => (match right
      | List.nil => (ctor Order.Greater)

definition{
  name: insertRow,
  parameters: [path : (List Nat), bytes : (List Nat), rows : Rows],
  result: Sorted,
  body: (match rows
    | Rows.Nil => (ctor Sorted.Canonical (ctor Rows.Cons path bytes (ctor Rows.Nil)))
    | Rows.Cons otherPath otherBytes more => (match (compareOctets path otherPath)
      | Order.Less => (ctor Sorted.Canonical (ctor Rows.Cons path bytes (ctor Rows.Cons otherPath otherBytes more)))
      | Order.Equal => (ctor Sorted.DuplicatePath)
      | Order.Greater => (match (insertRow path bytes more)
        | Sorted.DuplicatePath => (ctor Sorted.DuplicatePath)
        | Sorted.Canonical inserted => (ctor Sorted.Canonical (ctor Rows.Cons otherPath otherBytes inserted))))),
  recursive_argument: rows
}

definition{
  name: canonicalize,
  parameters: [rows : Rows],
  result: Sorted,
  body: (match rows
    | Rows.Nil => (ctor Sorted.Canonical (ctor Rows.Nil))
    | Rows.Cons path bytes more => (match (canonicalize more)
      | Sorted.DuplicatePath => (ctor Sorted.DuplicatePath)
      | Sorted.Canonical sorted => (insertRow path bytes sorted))),
  recursive_argument: rows
}

definition{
  name: isCanonical,
  parameters: [rows : Rows],
  result: Bool,
  body: (match rows
    | Rows.Nil => true
    | Rows.Cons path bytes more => (match more
      | Rows.Nil => true
      | Rows.Cons nextPath nextBytes rest => (match (compareOctets path nextPath)
        | Order.Less => (isCanonical more)
        | Order.Equal => false
        | Order.Greater => false))),
  recursive_argument: rows
}

definition{
  name: profileFraming,
  parameters: [files : (List (Bytes * Bytes)), reviews : (List Bytes), toolchain : Bytes],
  result: Framing,
  body: (match (canonicalize (rowsOfFiles files))
    | Sorted.DuplicatePath => (ctor Framing.DuplicatePath)
    | Sorted.Canonical rows => (match (canonicalizeReviews (recordsOfBytes reviews))
      | ReviewsSorted.DuplicateReview => (ctor Framing.DuplicateReview)
      | ReviewsSorted.Canonical sortedReviews => (match (toFit (sizeProfile (profileDomain) rows sortedReviews (octetsOfBytes toolchain)))
        | Fit.Within => (ctor Framing.Preimage (encProfile (profileDomain) rows sortedReviews (octetsOfBytes toolchain)))
        | Fit.Over => (ctor Framing.FieldTooLong)))),
  axioms: [Quot.sound, propext]
}

definition{
  name: identify,
  parameters: [framing : Framing],
  result: Outcome,
  body: (match framing
    | Framing.Preimage preimage => (ctor Outcome.Identified (Sha256::sha256 (bytesOfOctets preimage)))
    | Framing.DuplicatePath => (ctor Outcome.DuplicatePath)
    | Framing.DuplicateReview => (ctor Outcome.DuplicateReview)
    | Framing.FieldTooLong => (ctor Outcome.FieldTooLong)),
  axioms: [Quot.sound, propext]
}

definition{
  name: profileId,

definition{
  name: framedAlike,
  parameters: [left : Framing, right : Framing],
  result: Bool,
  body: (match left
    | Framing.Preimage found => (match right
      | Framing.Preimage wanted => (octetsEqual found wanted)
      | Framing.DuplicatePath => false
      | Framing.DuplicateReview => false
      | Framing.FieldTooLong => false)
    | Framing.DuplicatePath => false
    | Framing.DuplicateReview => false
    | Framing.FieldTooLong => false)
}

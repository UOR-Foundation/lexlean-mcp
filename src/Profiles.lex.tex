definition{
  name: underRoot,
  parameters: [root : Bytes, path : Bytes],
  result: Bool,
  body: (match (Identity::stripPrefix (Identity::octetsOfBytes root) (Identity::octetsOfBytes path))
    | Option.none => false
    | Option.some rest => (match rest
      | List.nil => false
      | List.cons c more => (c =? (separator)))),
  axioms: [Quot.sound, propext]
}

definition{
  name: insideRoots,
  parameters: [roots : (List Bytes), path : Bytes],
  result: Bool,
  body: (match roots
    | List.nil => false
    | List.cons root more => (if (underRoot root path) then true else (insideRoots more path))),
  axioms: [Quot.sound, propext],
  recursive_argument: roots
}

definition{
  name: profileFiles,
  parameters: [roots : (List Bytes), files : (List (Bytes * Bytes))],
  result: (List (Bytes * Bytes)),
  body: (match files
    | List.nil => (nil (Bytes * Bytes))
    | List.cons file rest => (if (insideRoots roots (first file)) then (profileFiles roots rest) else (cons file (profileFiles roots rest)))),
  axioms: [Quot.sound, propext],
  recursive_argument: files
}
